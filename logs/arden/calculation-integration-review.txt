"""Conservative tuition-award matching. No database writes or text inference."""

from datetime import date
from decimal import Decimal, InvalidOperation, ROUND_HALF_UP


def evaluate_award(
    award,
    rule,
    context,
    fee,
    currency,
    academic_year,
    study_year,
    today=None,
):
    today = today or date.today()

    if (
        not isinstance(rule, dict)
        or rule.get("verified") is not True
        or not rule.get("source_url")
    ):
        return None, "Rules have not been verified against a source"

    rules = rule.get("rules", {})

    required = {
        "benefit",
        "kind",
        "value",
        "academic_year",
        "study_years",
        "conditions",
        "award_status",
        "stacking",
        "compatible_ids",
        "percentage_basis",
        "fee_basis_confirmed",
    }

    if not isinstance(rules, dict) or not required.issubset(rules):
        return None, "Incomplete structured rules"

    if rules["fee_basis_confirmed"] is not True:
        return None, "Applicable tuition base has not been confirmed"

    if rules["benefit"] != "tuition":
        return None, "Award is not a tuition discount"

    if (
        not isinstance(rules["study_years"], list)
        or rules["academic_year"] != academic_year
        or study_year not in rules["study_years"]
    ):
        return None, "Award period does not match the fee period"

    if award.get("deadline") and today > award["deadline"]:
        return None, "Application deadline has passed"

    conditions = rules["conditions"]

    if not isinstance(conditions, dict):
        return None, "Conditions must be an object"

    # Unknown answers are never treated as confirmed eligibility.
    for key, allowed in conditions.items():
        if key not in context or context[key] is None:
            return None, f"Eligibility fact not confirmed: {key}"

        if not isinstance(allowed, list) or not allowed:
            return None, f"Invalid eligibility rule: {key}"

        if context[key] not in allowed:
            return None, f"Eligibility condition not met: {key}"

    if rules["award_status"] not in (
        "automatic",
        "competitive",
        "conditional",
    ):
        return None, "Unknown award status"

    if (
        rules["stacking"] not in ("exclusive", "explicit")
        or not isinstance(rules["compatible_ids"], list)
    ):
        return None, "Combination rules have not been confirmed"

    try:
        fee = Decimal(str(fee))
        value = Decimal(str(rules["value"]))

        if (
            not fee.is_finite()
            or fee < 0
            or not value.is_finite()
            or value <= 0
        ):
            raise ValueError()

        if rules["kind"] in ("fixed", "fixed_maximum"):
            if rules.get("currency") != currency:
                return None, "Award currency differs from fee currency"

            amount = value

        elif rules["kind"] == "percentage":
            if (
                value > 100
                or rules["percentage_basis"] != "original_tuition"
            ):
                return None, (
                    "Percentage basis is unsupported or unconfirmed"
                )

            amount = fee * value / Decimal("100")

        else:
            return None, "Unsupported award type"

    except (InvalidOperation, ValueError, TypeError):
        return None, "Invalid award amount"

    return {
        "id": str(award["id"]),
        "name": award["name"],
        "amount": min(fee, amount),
        "status": rules["award_status"],
        "stacking": rules["stacking"],
        "compatible_ids": [
            str(value) for value in rules["compatible_ids"]
        ],
        "is_maximum": rules["kind"] == "fixed_maximum",
    }, None


def select_awards(candidates, fee):
    """Select the best permitted combination of tuition awards."""

    if len(candidates) > 15:
        # Avoid exponential work for unusually large candidate lists.
        return [
            max(candidates, key=lambda candidate: candidate["amount"])
        ], True

    best = []
    best_total = Decimal("0")

    for mask in range(1, 1 << len(candidates)):
        group = [
            candidate
            for index, candidate in enumerate(candidates)
            if mask & (1 << index)
        ]

        # Every pair must explicitly permit combining with each other.
        if len(group) > 1 and any(
            first["stacking"] != "explicit"
            or second["stacking"] != "explicit"
            or second["id"] not in first["compatible_ids"]
            or first["id"] not in second["compatible_ids"]
            for index, first in enumerate(group)
            for second in group[index + 1:]
        ):
            continue

        total = min(
            fee,
            sum(
                (candidate["amount"] for candidate in group),
                Decimal("0"),
            ),
        )

        if total > best_total:
            best = group
            best_total = total

    return best, False


def calculate_scholarships(
    tuition,
    scholarships,
    profile,
    academic_year,
    study_year,
    allow_unverified_legacy=True,
):
    fee = Decimal(str(tuition.tuition_fee))
    legacy = Decimal(str(tuition.scholarship or 0))

    if (
        not fee.is_finite()
        or fee < 0
        or not legacy.is_finite()
        or legacy < 0
    ):
        raise ValueError(
            "Tuition and stored discount must be finite "
            "non-negative amounts"
        )

    linked_id = getattr(tuition, "scholarship_source_id", None)

    context = dict(
        profile,
        course_id=str(tuition.course.course_id),
        level=tuition.course.level,
        fee_status=tuition.residency,
    )

    review = []
    notes = []
    candidates = []
    scholarships = list(scholarships)

    # Preserve unidentified stored deductions, without stacking new awards.
    if legacy > 0 and not linked_id and allow_unverified_legacy:
        discount = min(fee, legacy)

        return {
            "discount": str(discount),
            "net": str(max(Decimal("0"), fee - discount)),
            "label": (
                "Net tuition with stored discount "
                "(eligibility unverified)"
            ),
            "applied": [],
            "review": [],
            "notes": [
                "A stored discount was deducted. Its eligibility "
                "has not been verified; additional awards "
                "were not evaluated."
            ],
            "maximum_award_estimate": False,
            "academic_year": academic_year,
            "study_year": study_year,
        }

    # Linking identifies the legacy award; it must not exclude other awards.
    if linked_id and not any(
        scholarship.pk == linked_id
        for scholarship in scholarships
    ):
        notes.append(
            "The linked scholarship was not found "
            "in the candidate list."
        )

    if legacy > 0 and not linked_id and not allow_unverified_legacy:
        notes.append(
            "An unverified stored discount was excluded; "
            "only verified award rules were evaluated."
        )

    for scholarship in scholarships:
        try:
            rule = scholarship.calculation_rule

            rule_data = {
                "verified": rule.verified,
                "source_url": rule.source_url,
                "rules": rule.rules,
            }

        except AttributeError:
            rule_data = None

        candidate, reason = evaluate_award(
            {
                "id": scholarship.pk,
                "name": scholarship.name,
                "deadline": scholarship.deadline,
            },
            rule_data,
            context,
            fee,
            tuition.currency,
            academic_year,
            study_year,
        )

        if candidate:
            candidates.append(candidate)
        else:
            review.append({
                "id": str(scholarship.pk),
                "reason": reason,
            })

    selected, limited = select_awards(candidates, fee)

    if limited:
        notes.append(
            "More than 15 eligible awards: "
            "selected the best single award."
        )

    discount = min(
        fee,
        sum(
            (candidate["amount"] for candidate in selected),
            Decimal("0"),
        ),
    ).quantize(
        Decimal("0.01"),
        rounding=ROUND_HALF_UP,
    )

    maximum_estimate = any(
        candidate.get("is_maximum", False)
        for candidate in selected
    )

    estimated = any(
        candidate["status"] != "automatic"
        for candidate in selected
    )

    if maximum_estimate:
        label = (
            "Estimated net tuition if maximum scholarship awarded"
        )

        notes.append(
            "This uses the published maximum annual award, "
            "not a guaranteed deduction. The university must "
            "confirm the award and amount."
        )

    elif estimated:
        label = "Estimated net tuition if awarded"

    elif selected:
        label = "Net tuition after automatic awards"

    else:
        label = "Tuition before unconfirmed awards"

    return {
        "discount": str(discount),
        "net": str(max(Decimal("0"), fee - discount)),
        "label": label,
        "maximum_award_estimate": maximum_estimate,
        "applied": [
            {
                "id": candidate["id"],
                "name": candidate["name"],
                "amount": str(candidate["amount"]),
                "status": candidate["status"],
                "is_maximum": candidate.get("is_maximum", False),
            }
            for candidate in selected
        ],
        "review": review,
        "notes": notes,
        "academic_year": academic_year,
        "study_year": study_year,
    }

def compare_tuition(request):
    """Compare tuition, applicable scholarships and deposit policies."""

    residence = request.GET.get("residence", "").strip().upper()
    destination = request.GET.get("destination", "").strip().upper()
    programme = request.GET.get("programme", "").strip().upper()
    subject = " ".join(request.GET.get("subject", "").split())

    fee_status = request.GET.get(
        "fee_status", "international"
    ).strip().lower()

    if not all([destination, programme, subject]):
        return JsonResponse(
            {"error": "Missing required parameters."},
            status=400,
        )

    country = {
        "UK": "UK",
        "IRELAND": "IE",
        "IE": "IE",
    }.get(destination)

    level = {
        "UNDERGRADUATE": "UG",
        "POSTGRADUATE": "PG",
        "UG": "UG",
        "PG": "PG",
        "PGT": "PG",
        "TOPUP": "TOPUP",
        "TOP-UP": "TOPUP",
        "MRES": "MRES",
    }.get(programme)

    if country is None or level is None:
        return JsonResponse(
            {"error": "Invalid destination or programme type."},
            status=400,
        )

    if fee_status != "international":
        return JsonResponse(
            {
                "error": (
                    "This comparison currently supports "
                    "international fee status only."
                )
            },
            status=400,
        )

    levels = {
        "UG": ["UG", "TOPUP"],
        "PG": ["PG", "PGT", "MRES"],
        "TOPUP": ["TOPUP"],
        "MRES": ["MRES"],
    }

    def parse_boolean(value):
        value = value.strip().lower()

        if value in {"yes", "true", "1"}:
            return True

        if value in {"no", "false", "0"}:
            return False

        return None

    profile = {"residence": residence}

    boolean_fields = (
        # General scholarship eligibility.
        "new_entrant",
        "full_time",
        "on_campus_dundee",
        "academic_conditions_met",
        "partner_discount",
        "repeat_year",
        "pg_2_1_equivalent_met",
        "offer_received",
        "applied_by_deadline",
        "external_funding",

        # Visa, funding and payment eligibility.
        "requires_student_visa",
        "self_funded",
        "full_fee_paying",
        "sponsored",

        # ARU eligibility.
        "apcl",
        "registration_conditions_met",
        "aru_funding_eligibility_confirmed",
        "aru_ug_award_confirmed",
        "aru_graduate",
        "aru_pg_entry_requirements_exceeded",
        "aru_pg_eligibility_confirmed",
        "aru_course_award_applicability_confirmed",
        "aru_international_route_eligibility_confirmed",
        "aru_london_alumni",
        "uk_first_class_degree",
        "aru_london_award_confirmed",
    )

    for key in boolean_fields:
        raw_answer = request.GET.get(key)

        # An omitted answer remains unknown.
        if raw_answer is None or not raw_answer.strip():
            continue

        answer = parse_boolean(raw_answer)

        if answer is None:
            return JsonResponse(
                {
                    "error": (
                        f"Invalid answer for {key}. "
                        "Use yes or no, or omit an unknown answer."
                    )
                },
                status=400,
            )

        profile[key] = answer

    entry_year = request.GET.get("entry_year", "").strip()

    if entry_year:
        try:
            entry_year_number = int(entry_year)
        except ValueError:
            return JsonResponse(
                {"error": "entry_year must be an integer."},
                status=400,
            )

        if not 1 <= entry_year_number <= 6:
            return JsonResponse(
                {"error": "entry_year must be between 1 and 6."},
                status=400,
            )

        profile["entry_year"] = entry_year_number

    award_tier = request.GET.get(
        "aru_ug_award_tier", ""
    ).strip().lower()

    if award_tier:
        if award_tier not in {"lower", "higher"}:
            return JsonResponse(
                {
                    "error": (
                        "aru_ug_award_tier must be lower or higher. "
                        "Omit it when the tier is unknown."
                    )
                },
                status=400,
            )

        profile["aru_ug_award_tier"] = award_tier

    residence_region = request.GET.get(
        "residence_region", ""
    ).strip()

    def merge_questions(*groups):
        """Preserve saved question definitions without duplicate keys."""
        merged = {}

        for group in groups:
            if not isinstance(group, list):
                continue

            for question in group:
                if (
                    isinstance(question, dict)
                    and question.get("key")
                ):
                    merged.setdefault(question["key"], question)

        return list(merged.values())

    def review_questions(raw, review_name):
        if not isinstance(raw, dict):
            return []

        review = raw.get(review_name)

        if not isinstance(review, dict):
            return []

        questions = review.get("eligibility_questions", [])

        return questions if isinstance(questions, list) else []

    def saved_scholarship_questions(
        awards,
        course_id,
        academic_year,
    ):
        """Return saved questions, filling gaps from rule conditions."""

        labels = {
            "new_entrant": (
                "Are you joining this university as a new student?"
            ),
            "full_time": "Will you study full-time?",
            "offer_received": (
                "Have you received an offer for this course?"
            ),
            "external_funding": (
                "Will you receive external funding for your studies?"
            ),
            "applied_by_deadline": (
                "Did you apply by the scholarship's published deadline?"
            ),
            "academic_conditions_met": (
                "Have you met the scholarship's academic conditions?"
            ),
            "entry_year": "Which course year will you enter?",
            "apcl": (
                "Have you received Accredited Prior Certified Learning?"
            ),
            "registration_conditions_met": (
                "Have all university registration conditions been met?"
            ),
            "aru_ug_award_confirmed": (
                "Has ARU confirmed your undergraduate scholarship?"
            ),
            "aru_ug_award_tier": (
                "Which undergraduate scholarship tier has ARU confirmed?"
            ),
            "aru_graduate": "Are you an ARU graduate?",
            "aru_pg_entry_requirements_exceeded": (
                "Has ARU confirmed that your qualifications exceed "
                "the scholarship's minimum entry requirements?"
            ),
            "aru_pg_eligibility_confirmed": (
                "Has ARU confirmed your postgraduate "
                "scholarship eligibility?"
            ),
            "aru_funding_eligibility_confirmed": (
                "Has ARU confirmed that your funding arrangement "
                "is eligible for this scholarship?"
            ),
        }

        # These facts come from the selected course or fee record.
        automatic_fields = {
            "course_id",
            "university_id",
            "level",
            "fee_status",
            "currency",
            "course_kind",
            "postgraduate_taught",
        }

        questions = {}

        for award in awards:
            try:
                rule = award.calculation_rule
            except AttributeError:
                continue

            data = rule.rules

            if (
                not rule.verified
                or not rule.source_url
                or not isinstance(data, dict)
                or data.get("academic_year") != academic_year
            ):
                continue

            study_years = data.get("study_years")

            # The comparison currently calculates study year 1.
            if (
                not isinstance(study_years, list)
                or 1 not in study_years
            ):
                continue

            conditions = data.get("conditions")

            if not isinstance(conditions, dict):
                continue

            scope = conditions.get("course_id")

            if scope is not None:
                if not isinstance(scope, list):
                    continue

                if str(course_id) not in {
                    str(value) for value in scope
                }:
                    continue

            saved = data.get("eligibility_questions", [])

            if not isinstance(saved, list):
                saved = []

            # Preserve explicit question definitions.
            for question in saved:
                if (
                    isinstance(question, dict)
                    and question.get("key")
                    and question["key"] not in automatic_fields
                ):
                    questions.setdefault(
                        question["key"],
                        dict(question),
                    )

            # Build missing definitions from actual saved conditions.
            for key, allowed in conditions.items():
                if (
                    key in automatic_fields
                    or key in questions
                    or not isinstance(allowed, list)
                    or not allowed
                ):
                    continue

                question = {
                    "key": key,
                    "question": labels.get(
                        key,
                        (
                            "Please provide your answer for "
                            f"{key.replace('_', ' ')}."
                        ),
                    ),
                }

                if all(type(value) is bool for value in allowed):
                    # Both yes and no must remain possible answers.
                    question["type"] = "boolean"

                elif all(type(value) is int for value in allowed):
                    question["type"] = "integer"

                elif all(isinstance(value, str) for value in allowed):
                    question["type"] = "choice"
                    question["choices"] = [
                        {
                            "value": value,
                            "label": value.replace("_", " ").capitalize(),
                        }
                        for value in allowed
                    ]
                    question["note"] = (
                        "These are the values covered by this rule. "
                        "Leave unanswered if unknown; do not select "
                        "an inaccurate answer."
                    )

                else:
                    question["type"] = "text"

                questions[key] = question

        return list(questions.values())
    def reviewed_fee_error(tuition, source):
        """Verify Aberystwyth and ARU fees before any deduction."""
        if source is None:
            return (
                "International tuition has not been verified "
                "for this course and intake."
            )

        raw = source.raw_fees

        if not isinstance(raw, dict):
            return "The saved tuition evidence requires review."

        display_notes = raw.get("display_notes")
        if not isinstance(display_notes, dict):
            display_notes = {}

        fee_review = raw.get("fee_review")
        if not isinstance(fee_review, dict):
            fee_review = {}

        reason = (
            raw.get("verification_reason")
            or display_notes.get("fee")
            or fee_review.get("reason")
            or (
                "International tuition verification is pending "
                "for this course and intake."
            )
        )

        if (
            raw.get("verified") is not True
            or not source.academic_year
            or not source.source_url
        ):
            return reason

        if source.fee_basis not in {
            "annual_tuition",
            "first_year_tuition",
            "full_time_course_fee",
        }:
            return (
                "A supported verified tuition amount is unavailable."
            )

        if (
            raw.get("academic_year") != source.academic_year
            or raw.get("currency") != tuition.currency
            or raw.get("source_fee_category") != "International"
        ):
            return (
                "The tuition source period, currency or "
                "fee category requires review."
            )

        try:
            stored = Decimal(str(tuition.tuition_fee))
            published = Decimal(str(raw["annual_tuition"]))

            if (
                not stored.is_finite()
                or not published.is_finite()
                or stored <= 0
                or published <= 0
                or stored != published
            ):
                return (
                    "Stored tuition does not match its "
                    "verified source amount."
                )

        except (
            KeyError,
            InvalidOperation,
            ValueError,
            TypeError,
        ):
            return "The published tuition amount requires review."

        return None

    courses = (
        Course.objects
        .annotate(comparison_search_title=Lower("title"))
        .filter(
            university__country=country,
            is_active=True,
            level__in=levels[level],
            comparison_search_title__contains=subject.lower(),
        )
        .select_related("university", "subject")
        .prefetch_related(
            "tuitions",
            "tuitions__source_metadata",
            "university__scholarships__calculation_rule",
        )
    )

    results = []

    for course in courses:
        university = course.university
        university_id = str(course.university_id)

        abertay = country == "UK" and university_id == "UK001"
        aberystwyth = country == "UK" and university_id == "UK002"
        aru = country == "UK" and university_id == "UK003"

        eligibility_questions = {
            "scholarship": [],
            "deposit": [],
        }

        base = {
            "university_id": course.university_id,
            "university_name": university.name,
            "university_slug": university.slug,
            "course_title": course.title,
            "course_id": course.course_id,
            "course_level": course.level,
            "duration": course.duration,
            "location": course.location,
            "start_date": (
                f"{course.start_month or ''} "
                f"{course.start_year or ''}"
            ).strip(),
            "course_url": course.course_url,
            "fee_status": "international",
        }

        def unavailable(reason):
            return {
                **base,
                "fee_available": False,
                "fee_verified": False,
                "fee_verification_status": "needs_review",
                "fee_label": "Fee verification required",
                "fee_note": reason,
                "academic_year": "",
                "currency": university.currency,
                "source_fee_category": None,
                "fee_source_url": "",
                "tuition_fee": None,
                "scholarship": None,
                "net_tuition": None,
                "scholarship_calculation": None,
                "tuition_deposit": None,
                "deposit_calculation": None,
                "eligibility_questions": eligibility_questions,
                "scholarship_info": (
                    "No deduction calculated without a confirmed fee."
                ),
            }

        tuitions = [
            item
            for item in course.tuitions.all()
            if item.residency == "INTL"
        ]

        if len(tuitions) != 1:
            results.append(
                unavailable(
                    "A unique international tuition record "
                    "is unavailable."
                )
            )
            continue

        tuition = tuitions[0]

        try:
            source = tuition.source_metadata
        except TuitionSource.DoesNotExist:
            source = None

        raw = (
            source.raw_fees
            if source is not None
            and isinstance(source.raw_fees, dict)
            else {}
        )

        # Preserve pending-route questions even when the fee is unavailable.
        eligibility_questions = {
            "scholarship": review_questions(
                raw, "scholarship_review"
            ),
            "deposit": review_questions(raw, "deposit_review"),
        }

        if abertay:
            reason = abertay_fee_verification_error(tuition, source)

            if reason:
                results.append(unavailable(reason))
                continue

        if aberystwyth or aru:
            reason = reviewed_fee_error(tuition, source)

            if reason:
                results.append(unavailable(reason))
                continue

        try:
            amount = Decimal(str(tuition.tuition_fee))
        except (InvalidOperation, TypeError, ValueError):
            results.append(
                unavailable("The stored tuition amount is invalid.")
            )
            continue

        if not amount.is_finite() or amount <= 0:
            results.append(
                unavailable("The stored tuition amount is invalid.")
            )
            continue

        metadata = dict(fee_metadata(tuition) or {})

        academic_year = (
            source.academic_year

            else metadata.get("academic_year")
        )

        if not academic_year:
            if course.start_year is None:

                    unavailable("The applicable fee year is missing.")
                )
                continue

            # Fallback for universities not yet covered by reviewed adapters.
            academic_year = (
                f"{course.start_year}/"
                f"{(course.start_year + 1) % 100:02d}"
            )

        currency = tuition.currency
        fee_source_url = source.source_url if source else ""

        if abertay:

            if not isinstance(categories, list):
                categories = []

            category = next(
                (
                    item

                    if isinstance(item, dict)

                    == "International"
                    and item.get("academic_year") == academic_year
                ),
                {},
            )

            note = (
                "Published international fee; the university "
                "confirms individual eligibility."
            )

            annotation = category.get("annotation")

            if isinstance(annotation, str) and annotation.strip():
                note += " " + annotation.strip()

            metadata = {
                "fee_label": "Annual tuition fee",
                "academic_year": academic_year,
                "fee_note": note,
            }

        elif aberystwyth or aru:
            metadata = {
                **metadata,
                "fee_label": {
                    "annual_tuition": "Annual tuition fee",
                    "first_year_tuition": "First-year tuition fee",
                    "full_time_course_fee": (
                        "Published full-time course fee"
                    ),
                }.get(source.fee_basis, "Published tuition fee"),
                "academic_year": academic_year,
                "fee_note": (
                    raw.get("fee_note")
                    or (
                        "Published international tuition for the "
                        "reviewed course and fee period. The university "
                        "confirms individual eligibility."
                    )
                ),
            }

        award_level = {
            "PGT": "PG",
            "TOPUP": "UG",
        }.get(course.level, course.level)

        awards = [
            award
            for award in university.scholarships.all()
            if award.level == award_level
        ]

        comparison_tuition = SimpleNamespace(
            tuition_fee=amount,
            scholarship=tuition.scholarship,
            scholarship_source_id=getattr(
                tuition, "scholarship_source_id", None
            ),
            currency=currency,
            residency="INTL",
            course=course,
        )

        course_profile = dict(profile)

        if abertay:
            course_profile["postgraduate_taught"] = (
                raw.get("course_kind") == "postgraduate_taught"
            )

        # The calculator matches the course, fee year, study year,
        # eligibility conditions and permitted award combinations.
        # Unknown personal facts are not automatically set to True.
        calculation = calculate_scholarships(
            comparison_tuition,
            awards,
            course_profile,
            academic_year=academic_year,
            study_year=1,
            allow_unverified_legacy=(
                university_id not in {"UK001", "UK002", "UK003"}
            ),
        )

        discount = Decimal(str(calculation["discount"]))
        net_tuition = Decimal(str(calculation["net"]))

        scholarship_info = "; ".join(
            (
                f"{award['name']}: {currency} "
                f"{'up to ' if award.get('is_maximum') else ''}"
                f"{award['amount']} "
                "(for the compared study year)"
            )
            for award in calculation.get("applied", [])
        )

        if not scholarship_info:
            scholarship_info = (
                "Stored discount applied; eligibility unverified"
                if discount > 0
                else "No verified award applied"
            )

        deposit_calculation = {
            "available": False,
            "amount": None,
            "currency": currency,
            "academic_year": academic_year,
            "source_url": None,
            "policy_checked_at": None,
            "rule": None,
            "is_estimate": True,
            "label": "Deposit assessment required",
            "note": (
                "Deposit rules have not been verified "
                "for this university."
            ),
            "missing_facts": [],
        }

        if abertay:
            deposit_calculation = calculate_abertay_deposit(
                academic_year=academic_year,
                tuition_fee=amount,
                scholarship_calculation=calculation,
                legal_residence=residence,
                residence_region=residence_region,
                requires_student_visa=profile.get(
                    "requires_student_visa"
                ),
                self_funded=profile.get("self_funded"),
            )

        elif aberystwyth:
            deposit_calculation = calculate_aberystwyth_deposit(
                academic_year=academic_year,
            )

        elif aru:
            deposit_calculation = calculate_aru_deposit(
                tuition=tuition,
                academic_year=academic_year,
                profile=course_profile,
            )

            aru_questions = get_aru_eligibility_questions(
                tuition=tuition,
                academic_year=academic_year,
            ) or {}

            eligibility_questions = {
                "scholarship": merge_questions(
                    eligibility_questions["scholarship"],
                    aru_questions.get("scholarship", []),
                ),
                "deposit": merge_questions(
                    eligibility_questions["deposit"],
                    aru_questions.get("deposit", []),
                ),
            }

        # Return saved scholarship questions for Aberystwyth too.
        eligibility_questions["scholarship"] = merge_questions(
            eligibility_questions["scholarship"],
            saved_scholarship_questions(
                awards,
                course.course_id,
                academic_year,
            ),
        )

        verified_fee = abertay or aberystwyth or aru

        results.append({
            **base,
            **metadata,
            "academic_year": academic_year,
            "fee_available": True,
            "fee_verified": verified_fee,
            "fee_verification_status": (
                "verified" if verified_fee else "not_checked"
            ),
            "source_fee_category": "International",
            "fee_source_url": fee_source_url,
            "currency": currency,

            # Gross tuition is retained separately.
            "tuition_fee": float(amount),

            # Applicable scholarship and resulting net tuition.
            "scholarship": float(discount),
            "net_tuition": float(net_tuition),
            "scholarship_calculation": calculation,
            "scholarship_info": scholarship_info,

            "tuition_deposit": (
                float(deposit_calculation["amount"])
                if deposit_calculation.get("available")
                and deposit_calculation.get("amount") is not None
                else None
            ),
            "deposit_calculation": deposit_calculation,
            "eligibility_questions": eligibility_questions,
        })

    results.sort(
        key=lambda result: (
            not result["fee_available"],
            result["currency"] or "",
            (
                result["net_tuition"]
                if result["net_tuition"] is not None
                else 0
            ),
        )
    )

    return JsonResponse({
        "results": results,
        "count": len(results),
        "filters": {
            "country": country,
            "level": level,
            "subject": subject,
            "fee_status": "international",
            "residency": "INTL",
        },
    })

"""Read ARU deposit policies and eligibility questions from the database."""

from copy import deepcopy
from decimal import Decimal, InvalidOperation

from app.models import DepositPolicy, ScholarshipRule


UNIVERSITY_ID = "UK003"


def boolean_answer(value):
    """Unknown answers stay unknown."""
    if value is True or value is False:
        return value

    if isinstance(value, str):
        value = value.strip().casefold()

        if value in {"yes", "true", "1"}:
            return True

        if value in {"no", "false", "0"}:
            return False

    return None


def saved_policy(academic_year):
    return DepositPolicy.objects.filter(
        university_id=UNIVERSITY_ID,
        academic_year=academic_year,
        currency="GBP",
        verified=True,
    ).first()


def get_aru_eligibility_questions(*, tuition, academic_year):
    """
    Return saved questions for this course and academic year.

    Does not infer answers or modify the database.
    """
    result = {
        "scholarship": [],
        "deposit": [],
    }

    if tuition.course.university_id != UNIVERSITY_ID:
        return result

    questions = {}
    allowed_tiers = set()

    rules = ScholarshipRule.objects.filter(
        scholarship__university_id=UNIVERSITY_ID,
        verified=True,
    )

    for saved_rule in rules:
        rule = saved_rule.rules

        if not isinstance(rule, dict):
            continue

        conditions = rule.get("conditions", {})

        if not isinstance(conditions, dict):
            continue

        if (
            rule.get("academic_year") != academic_year
            or str(tuition.course_id)
            not in conditions.get("course_id", [])
        ):
            continue

        allowed_tiers.update(
            conditions.get("aru_ug_award_tier", [])
        )

        for question in rule.get("eligibility_questions", []):
            if (
                isinstance(question, dict)
                and isinstance(question.get("key"), str)
            ):
                questions.setdefault(
                    question["key"], deepcopy(question)
                )

    tier_question = questions.get("aru_ug_award_tier")

    if tier_question is not None:
        tier_question["choices"] = [
            choice
            for choice in tier_question.get("choices", [])
            if choice.get("value") in allowed_tiers
        ]

    result["scholarship"] = list(questions.values())

    policy = saved_policy(academic_year)

    if policy and isinstance(policy.rules, dict):
        if str(tuition.course_id) in policy.rules.get(
            "course_ids", []
        ):
            result["deposit"] = deepcopy(
                policy.rules.get("eligibility_questions", [])
            )

    return result


def calculate_aru_deposit(
    *,
    tuition,
    academic_year,
    profile,
):
    """
    Calculate the fixed deposit from the saved ARU policy.

    The deposit is part of tuition, not an additional fee.
    Scholarship discounts are not subtracted from this fixed amount.
    """
    result = {
        "available": False,
        "amount": None,
        "currency": "GBP",
        "academic_year": academic_year,
        "source_url": None,
        "policy_checked_at": None,
        "rule": None,
        "is_estimate": True,
        "label": "Deposit assessment required",
        "note": "",
        "missing_facts": [],
    }

    def unavailable(note, missing=None):
        return {
            **result,
            "note": note,
            "missing_facts": missing or [],
        }

    if tuition.course.university_id != UNIVERSITY_ID:
        return unavailable("This calculation applies to ARU only.")

    if tuition.residency != "INTL" or tuition.currency != "GBP":
        return unavailable(
            "This policy applies to international tuition in GBP."
        )

    policy = saved_policy(academic_year)

    if policy is None:
        return unavailable(
            "No verified ARU deposit policy is saved "
            "for this academic year."
        )

    result["source_url"] = policy.source_url
    result["policy_checked_at"] = (
        policy.checked_at.isoformat()
        if policy.checked_at else None
    )

    rules = policy.rules

    if not isinstance(rules, dict):
        return unavailable("The saved deposit policy requires review.")

    if (
        rules.get("schema_version") != 1
        or rules.get("kind") != "fixed"
        or rules.get("currency") != "GBP"
        or rules.get("fee_status") != "INTL"
        or rules.get("payment_stage") != "before_cas"
        or rules.get("academic_year") != academic_year
        or rules.get("requires_student_visa") is not True
        or rules.get("full_fee_paying") is not True
        or rules.get("deduct_scholarship_from_deposit") is not False
    ):
        return unavailable("The saved deposit rule is unsupported.")

    scope = rules.get("course_ids")

    if not isinstance(scope, list) or str(tuition.course_id) not in scope:
        return unavailable(
            "This course has not been included in the "
            "reviewed deposit policy."
        )

    answers = {
        key: boolean_answer(profile.get(key))
        for key in (
            "requires_student_visa",
            "full_fee_paying",
            "sponsored",
        )
    }

    missing = [
        key for key, value in answers.items()
        if value is None
    ]

    if missing:
        return unavailable(
            "Confirm the visa, fee-paying and sponsorship "
            "answers to calculate the applicable deposit.",
            missing,
        )

    if not answers["requires_student_visa"]:
        return unavailable(
            "The pre-CAS rule does not apply without a Student "
            "visa requirement. Confirm payment arrangements with ARU."
        )

    if answers["sponsored"]:
        return unavailable(
            "A sponsorship letter may affect the payment requirement. "
            "ARU must confirm the deposit for this arrangement."
        )

    if not answers["full_fee_paying"]:
        return unavailable(
            "This published rule covers full-fee-paying international "
            "students. Confirm your payment arrangement with ARU."
        )

    try:
        amount = Decimal(str(rules["amount"]))

        if (
            not amount.is_finite()
            or amount <= 0
            or amount != amount.quantize(Decimal("0.01"))
        ):
            raise ValueError("Invalid deposit amount.")

    except (KeyError, InvalidOperation, ValueError, TypeError):
        return unavailable("The saved deposit amount requires review.")

    return {
        **result,
        "available": True,
        "amount": format(amount, ".2f"),
        "rule": "fixed_before_cas",
        "label": "Published deposit before CAS",
        "note": (
            f"The saved ARU policy specifies £{amount:,.2f} "
            "before CAS for this payment arrangement. "
            "Confirm the final requirement in your offer letter. "
            "This payment forms part of tuition; do not add it "
            "to the tuition total. Scholarship discounts have "
            "not been subtracted from the fixed deposit."
        ),
        "missing_facts": [],
    }

"""Calculate Abertay deposit estimates from saved database policy."""

from decimal import Decimal, InvalidOperation, ROUND_HALF_UP

from app.models import DepositPolicy


COUNTRY_ALIASES = {
    "NIGERIA": "NG",
    "AFGHANISTAN": "AF",
    "IRAN": "IR",
    "PAKISTAN": "PK",
    "BANGLADESH": "BD",
    "INDIA": "IN",
}


def money(value):
    amount = Decimal(str(value))

    if not amount.is_finite() or amount < 0:
        raise ValueError("Invalid monetary amount.")

    return amount.quantize(
        Decimal("0.01"),
        rounding=ROUND_HALF_UP,
    )


def percentage(value):
    amount = Decimal(str(value))

    if (
        not amount.is_finite()
        or amount <= 0
        or amount > 100
    ):
        raise ValueError("Invalid percentage.")

    return amount

def calculate_aberystwyth_deposit(*, academic_year):
    policy = DepositPolicy.objects.filter(
        university_id="UK002",
        academic_year=academic_year,
        currency="GBP",
        verified=True,
    ).first()

    result = {
        "available": False,
        "amount": None,
        "currency": "GBP",
        "academic_year": academic_year,
        "source_url": None,
        "policy_checked_at": None,
        "rule": None,
        "is_estimate": False,
        "label": "Deposit assessment required",
        "note": (
            "No verified deposit policy is saved for this academic year."
        ),
    }

    if policy is None:
        return result

    rules = policy.rules

    if (
        not isinstance(rules, dict)
        or rules.get("kind") != "offer_letter"
    ):
        return {
            **result,
            "note": "The saved deposit policy requires review.",
        }

    return {
        **result,
        "source_url": policy.source_url,
        "policy_checked_at": policy.checked_at.isoformat(),
        "rule": "offer_letter",
        "label": "Confirm deposit in your offer letter",
        "note": rules["note"],
    }

def calculate_abertay_deposit(
    *,
    academic_year,
    tuition_fee,
    scholarship_calculation,
    legal_residence,
    residence_region="",
    requires_student_visa=None,
    self_funded=None,
):
    result = {
        "available": False,
        "amount": None,
        "currency": "GBP",
        "academic_year": academic_year,
        "source_url": None,
        "policy_checked_at": None,
        "rule": None,
        "is_estimate": True,
        "label": "Deposit assessment required",
        "note": "",
    }

    def unavailable(reason):
        return {**result, "note": reason}

    policy = DepositPolicy.objects.filter(
        university_id="UK001",
        academic_year=academic_year,
        verified=True,
        currency="GBP",
    ).first()

    if policy is None:
        return unavailable(
            "No verified deposit policy is saved for this academic year."
        )

    result["source_url"] = policy.source_url
    result["policy_checked_at"] = policy.checked_at.isoformat()

    rules = policy.rules

    if not isinstance(rules, dict):
        return unavailable("Saved deposit policy requires review.")

    if (
        rules.get("schema_version") != 1
        or rules.get("fee_status") != "INTL"
        or rules.get("payment_stage") != "before_cas"
        or rules.get("basis") != "net_tuition"
        or rules.get("requires_student_visa") is not True
        or rules.get("self_funded") is not True
    ):
        return unavailable("Saved deposit policy is unsupported.")
    residence = (legal_residence or "").strip().upper()
    residence = COUNTRY_ALIASES.get(residence, residence)

    if not residence:
        return {
            **result,
            "label": "General deposit guidance",
            "note": (
                f"Normally {rules['standard_percentage']}% of net "
                "tuition; country-specific exceptions apply. "
                "Select your legal residence and confirm your "
                "visa and funding status for an estimate."
            ),
    }
    if requires_student_visa is None:
        return unavailable(
            "Confirm whether the student requires a Student visa."
        )

    if requires_student_visa is not True:
        return unavailable(
            "The pre-CAS deposit rule does not apply to students "
            "who do not require a Student visa. "
            "Confirm payment arrangements with Abertay."
        )

    if self_funded is not True:
        return unavailable(
            "Confirm that the student is self-funded. "
            "Sponsored students require separate payment assessment."
        )

    residence = (legal_residence or "").strip().upper()
    residence = COUNTRY_ALIASES.get(residence, residence)

    if not re_country_code(residence):
        return unavailable(
            "Select a specific legal country of residence."
        )

    region = " ".join(
        (residence_region or "").casefold().split()
    )

    region_rules = rules.get("full_payment_regions")
    fixed_rules = rules.get("country_fixed_amounts")
    full_countries = rules.get("full_payment_countries")

    if (
        not isinstance(region_rules, dict)
        or not isinstance(fixed_rules, dict)
        or not isinstance(full_countries, list)
        or not all(
            isinstance(values, list)
            for values in region_rules.values()
        )
    ):
        return unavailable("Saved country deposit rules require review.")

    if residence in region_rules and not region:
        return unavailable(
            "The state or region of legal residence is required "
            "because deposit rules differ by region."
        )

    if not isinstance(scholarship_calculation, dict):
        return unavailable("A tuition calculation is required.")

    try:
        gross = money(tuition_fee)
        discount = money(scholarship_calculation["discount"])
        net = money(scholarship_calculation["net"])

        if (
            gross <= 0
            or discount > gross
            or net != gross - discount
        ):
            raise ValueError("Inconsistent tuition calculation.")

        if residence in fixed_rules:
            deposit = money(fixed_rules[residence])
            selected_rule = "country_fixed_amount"

        elif (
            residence in full_countries
            or region in region_rules.get(residence, [])
        ):
            rate = percentage(rules["full_payment_percentage"])
            deposit = money(net * rate / Decimal("100"))
            selected_rule = "full_net_tuition"

        else:
            rate = percentage(rules["standard_percentage"])
            deposit = money(net * rate / Decimal("100"))
            selected_rule = "percentage_of_net_tuition"

        if deposit > net:
            return unavailable(
                "The published deposit exceeds the calculated net "
                "tuition. Abertay must confirm the amount payable."
            )

    except (
        InvalidOperation,
        ValueError,
        TypeError,
        KeyError,
    ):
        return unavailable(
            "The saved policy or tuition calculation requires review."
        )

    notes = [
        "Calculated from Abertay's saved deposit policy for a new, "
        "self-funded student requiring a Student visa.",
        "Abertay must confirm the final payment in the offer "
        "or CAS process.",
        "Early-payment discounts have not been included.",
    ]

    if rules.get("individual_override_possible") is True:
        notes.append(
            "Abertay may require a higher deposit after an "
            "individual visa-risk assessment."
        )

    if discount > 0:
        notes.append(
            "Any scholarship used must be awarded and confirmed by "
            "Abertay. Self-reported eligibility does not confirm an award."
        )

    return {
        **result,
        "available": True,
        "amount": str(deposit),
        "rule": selected_rule,
        "label": "Estimated deposit before CAS",
        "note": " ".join(notes),
    }


def re_country_code(value):
    return (
        len(value) == 2
        and value.isascii()
        and value.isalpha()
    )
